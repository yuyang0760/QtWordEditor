## 4. 详细设计

### 4.1 UnifiedCursor 类完整接口

```cpp
class UnifiedCursor : public QObject
{
    Q_OBJECT
public:
    explicit UnifiedCursor(DocumentView* view, QObject *parent = nullptr);
    ~UnifiedCursor() override;

    // ========== 光标移动 ==========
    void moveLeft();
    void moveRight();
    void moveUp();
    void moveDown();
    void moveToStartOfLine();
    void moveToEndOfLine();
    void moveToStartOfDocument();
    void moveToEndOfDocument();
    
    // ========== 光标定位 ==========
    void setPosition(ParagraphBlock* paragraph, int spanIndex, int charOffset);
    void setPosition(MathSpan* mathSpan, int childIndex);
    void setPositionFromScenePoint(const QPointF& scenePos);
    
    // ========== 编辑操作 ==========
    void insertText(const QString& text);
    void deleteChar();
    void backspace();
    
    // ========== 光标状态 ==========
    CursorState state() const;
    bool isInMath() const;
    
    // ========== 视图更新 ==========
    void updateCursor();
    QGraphicsLineItem* cursorItem() const;

signals:
    void positionChanged();
    void contentChanged();

private:
    TextBlockItem* getCurrentTextBlockItem() const;
    void initializeCursor();

private:
    DocumentView* m_view;
    CursorState m_state;
    QGraphicsLineItem* m_cursorItem;
};
```

### 4.2 CursorState 结构体

```cpp
struct CursorState {
    ParagraphBlock* paragraph;    // 当前段落
    int spanIndex;                // 当前 InlineSpan 索引
    int charOffset;               // 在 Span 内的字符偏移
    bool isMath;                  // 是否在公式内
    MathSpan* mathSpan;           // 当前公式 Span
    int mathChildIndex;           // 公式内子项索引
    
    CursorState()
        : paragraph(nullptr)
        , spanIndex(0)
        , charOffset(0)
        , isMath(false)
        , mathSpan(nullptr)
        , mathChildIndex(0)
    {}
};
```
